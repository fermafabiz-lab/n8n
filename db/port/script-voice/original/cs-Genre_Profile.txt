let p = PROFILES[tone] || FACTUAL;
let source = 'built-in';

// Prefer the Airtable row when it exists and is coherent.
try {
  const rec = ($input.first().json.records || [])[0];
  const f = rec && rec.fields;
  if (f && f['Format'] && f['Structure'] && f['Voice']) {
    p = {
      format: f['Format'],
      research: f['Research'] === true,
      researchBrief: f['Research Brief'] || '',
      factSheetFraming: f['Fact Sheet Framing'] || '',
      invention: f['Invention'] || '',
      structure: f['Structure'],
      voice: f['Voice'],
      wpm: Number(f['WPM']) || p.wpm,
      hookRule: f['Hook Rule'] || p.hookRule,
      montageIntensity: Number.isFinite(Number(f['Montage Intensity'])) ? Number(f['Montage Intensity']) : p.montageIntensity,
      visual: f['Visual'] || p.visual,
    };
    source = 'airtable';
  }
} catch (e) { /* fetch failed — built-in profile stands */ }

return [{ json: { p: Object.assign({}, p, { tone: $('Receive Project Data').first().json.Tonalitate || 'Documentary', profileSource: source }) } }];
